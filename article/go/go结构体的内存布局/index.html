测试环境：go1.25.3 windows/amd64

1. 结构体布局#

Go 语言中，结构体和它所包含的数据在内存中是以连续块的形式存在的，即使结构体中嵌套有其他的结构体。

2. 示例#

2.1. 结构体#

package main

import (
	"fmt"
	"reflect"
)

type studentInfo struct {
	id    int     // 学号
	name  string  // 姓名
	sorce float32 // 总分
	rank  int     // 年纪排名
}

func main() {
	var s studentInfo = studentInfo{
		id:    1001,
		name:  "张三",
		sorce: 670.5,
		rank:  1,
	}

	println("int 的大小:", reflect.TypeOf(int(0)).Size())
	println("float32 的大小:", reflect.TypeOf(float32(0)).Size())

	// 获取结构体的型信息
	var t reflect.Type = reflect.TypeOf(s)
	fmt.Println("结构体大小:", t.Size(), "字节")
	fmt.Println("结构体对齐要求:", t.Align(), "字节")
}

package main

import (
	"fmt"
	"unsafe"
)

type studentInfo struct {
	id    int     // 学号
	name  string  // 姓名
	sorce float32 // 总分
	rank  int     // 年纪排名
}

func main() {
	var s studentInfo = studentInfo{
		id:    1001,
		name:  "张三",
		sorce: 670.5,
		rank:  1,
	}

	// 拿到结构体的地址
	var ptr uintptr = uintptr(unsafe.Pointer(&s))

	// 打印这个结构体的所有内存信息
	fmt.Println("结构体地址存放的数值:")
	fmt.Printf("0x%x: ", ptr)
	for i := uintptr(0); i < unsafe.Sizeof(s); i++ {
		if (i%8 == 0) && (i != 0) {
			fmt.Printf("\n")
			fmt.Printf("0x%x: ", ptr+i)
		}
		var bytePtr *uint8 = (*uint8)(unsafe.Pointer(ptr + i))
		fmt.Printf("0x%02x ", *bytePtr)
	}
	fmt.Println()

	strPtr := unsafe.StringData(s.name)
	strLen := len(s.name)
	fmt.Printf("字符串指针: 0x%x\n", strPtr)
	fmt.Printf("字符串长度: %d\n", strLen)

	fmt.Print("UTF-8 编码: ")
	for i := 0; i < strLen; i++ {
		var bytePtr *uint8 = (*uint8)(unsafe.Pointer(uintptr(unsafe.Pointer(strPtr)) + uintptr(i)))
		fmt.Printf("%02x ", *bytePtr)
	}
	fmt.Println()
}

UTF-8 编码: e5 bc a0 e4 b8 89

  1. 0xc00005de98

    存放张三这个字符串的地址。

  2. 0xc00005dea0

    张 的 UTF-8 编码是 0xE5 0xBC 0xA0，三 的 UTF-8 编码是 0xE4 0xB8 0x89，所以总长度是 0x06。

  3. 0xc00005dea8

    670.5 的 IEEE 754 格式是：0x44 0x27 0x40 0x00。
